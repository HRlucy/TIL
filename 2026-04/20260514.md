## SQL

# select
> SELECT 컬럼명 FROM 테이블명; 특정 테이블에서 원하는 데이터를 조회하는데 사용

1. 단일 열(컬럼) 데이터 검색
* SELECT menu_name from tbl_menu;
2. 여러 열의 데이터 검색
* select menu_code, menu_name, menu_price from tbl_menu;
3. 모든 열에서 데이터 검색(* : 모든 컬럼을 의미하는 와일드카드)
* select * from tbl_menu;

# ORDER BY
1. 오름차순정렬
* select menu_code from tbl_menu order by menu_price asc;
2. 내림차순 정렬
* select menu_code from tbl_menu order by menu_price desc;

# WHERE 조건
> '조건 필터링'의 역할, 특정 조건에 맞는 레코드만을 선택하는데 사용
1. 비교 연산자 활용(=, <>, >, <=)
* where orderable_status <> 'Y';
* where menu_price < 13000;
2. 논리연산자(and, or)를 활용하여 조건 조합
* where orderable_status = 'Y' and category_code =10;
3. LIKE 조건 메뉴이름에 '마늘' 포함된 메뉴 찾고 싶을 때, %는 0개 이상의 모든문자를 의미하는 와일드 카드
* where menu_name like '%마늘%';
4. IN 연산자 카테고리 코드가 4번 or 5번 or 6번인것 제외
* where category_code in (4,5,6)
5. is null 연산자 데이터 베이스에서 null은 '0'이나 '빈 문자열'이 아닌, '값이 존재하지 않음'을 의미하는 상태
* where ref_category_code is null;

# distinct
> 중복된 값을 제거하고 유니크한 값들의 목록만 보고 싶을 때 사용 특정 컬럼에 어떤 값들이 종류별로 있는지 확인할 때 유용
* select distinct category_code from tbl_menu order by category_code;

# limit
> 결과중에 딱 내가 원하는 만큼만 잘라서 주라고 요청하는 키워드
* order by menu_price desc limit 5;

# join
1. 컬럼 별칭
* menu_name as '메뉴 이름'
2. 테이블 별칭(join에서 필수)
* tbl_menu as tm
3. inner join(교집합) 가장 기본적인 조인, inner 키워드 생략 가능
* join tbl_category tc on tm.category_code = tc.category_code;
4. outer join(left join, right join) : left : 조인구문을 기준으로 왼쪽 테이블의 데이터는 모두 보여주고, 오른쪽 테이블에서 짝이 맞는 데이터가 없으면 'null'로 채워 보여줌
* left join tbl_menu b on a.category_code =b.category_code; 

# group by
> 지정된 컬럼의 값이 같은 데이터들을 하나의 그룹으로 묶음
* group by category_code;

# having 
> 그룹에 대한 조건 필터링 where는 그룹화 하기 전, 원본 데이터에 대한 조건

# roll up
> 그룹별 집계 결과와 함께 그 그룹들의 총계를 보여줌
