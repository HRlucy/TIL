# 1급시민객체 함수

- 함수는 값으로써 처리될 수 있는 1급시민객체
- 변수에 대입할 수 있음
- 리턴 값으로써 사용될 수 있음

def foo():
  print('fooooooo')

print (foo, type(foo))

doo = foo
print(doo, type(doo))

def runner(f, n=10):
  """함수실행기"""
  for i in range(n):
    f()
runner(foo)

def get_func():
  def bar():
    print('baaaaaaaaar')
  return bar

f = get_func()

f()

# lambda 함수
한줄로 작성하는 익명 함수표현식

def square(x):
  return x **2
  
square(3)


# lambda 매개변수부 : 리턴값
f = lambda x: x ** 2
f(3)

add = lambda a, b=3: a+b
print(add(10, 20))
print(add(10))

# 1~10까지의 제곱수를 요소로 하는 리스트 생성
squares = []
for n in range(1, 11):
  squares.append(n ** 2)
print(squares)

squares2 = [square(n) for n in range(1, 11)]
print(squares2)

square3 = [(lambda x: x **2)(n) for n in range(1, 11)]
print(square3)

# 전달한 모든 정수의 합을 반환하는 lambda 함수 선언
# sum() builtins 함수
adder = lambda *args: sum(args)
print(adder(10, 23, 55, 10))
