# 파이선 numpy
import numpy as np

def n_shape():
    print("== shape ==")

    print(np.array([80, 90, 100]).shape)
    print(np.array([
        [80, 90, 100],
        [70, 88, 99]
    ]).shape)
    print(np.zeros((2, 3, 4)).shape)

def n_ndim():
    print("=== ndim ===")
    print(np.array([80, 90, 100]).ndim)
    print(np.array([
        [80, 90, 100],
        [70, 88, 99]
    ]).ndim)
    print(np.zeros((2, 3, 4)).ndim)

def n_size():
    print("== size ==")
    print(np.array([[80, 90, 100], [99, 24, 67]]).size)
    print(np.zeros((2, 3, 4)).size)

def n_dtype():
    print("=== dtype ===")
    print(np.array([10, 20, 30]).dtype)
    print(np.array([10., 20., 30.]).dtype)

    # 자동 문자열 변환 Numpy는 하나의 자료형만 저장 -> 섞이면 자동 변환
    data = np.array(([100, 200, "python"]))
    print(data)
    print(data.dtype)



if __name__ == "__main__":
    n_shape()
    n_ndim()
    n_size()
    n_dtype()
