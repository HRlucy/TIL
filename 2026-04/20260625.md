## 실전 프로젝트 sql 생성
CREATE EXTENSION IF NOT EXISTS pg_trgm;

CREATE TABLE IF NOT EXISTS product_categories (
    product_category_id BIGSERIAL PRIMARY KEY,
    name VARCHAR(100) UNIQUE NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NULL
);

CREATE TABLE IF NOT EXISTS products (
    product_id BIGSERIAL PRIMARY KEY,
    product_category_id BIGINT NOT NULL REFERENCES product_categories(product_category_id),
    name VARCHAR(100) UNIQUE NOT NULL,
    default_expiry_days INTEGER NULL,
    is_active BOOLEAN NOT NULL DEFAULT TRUE,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NULL
);

CREATE INDEX IF NOT EXISTS idx_products_name_trgm ON products USING gin (name gin_trgm_ops);

CREATE TABLE IF NOT EXISTS recipe_categories (
    recipe_category_id BIGSERIAL PRIMARY KEY,
    name VARCHAR(100) UNIQUE NOT NULL
);

CREATE TABLE IF NOT EXISTS recipes (
    recipe_id BIGSERIAL PRIMARY KEY,
    recipe_category_id BIGINT NOT NULL REFERENCES recipe_categories(recipe_category_id),
    name VARCHAR(30) NOT NULL,
    cooking_time INTEGER NOT NULL DEFAULT 15,
    difficulty VARCHAR(20) NOT NULL DEFAULT 'EASY',
    description VARCHAR(1000) NULL,
    is_deleted BOOLEAN NOT NULL DEFAULT FALSE,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NULL,
    created_by BIGINT NULL,
    updated_by BIGINT NULL
);

CREATE TABLE IF NOT EXISTS recipe_steps (
    recipe_step_id BIGSERIAL PRIMARY KEY,
    recipe_id BIGINT NOT NULL REFERENCES recipes(recipe_id),
    step_no INTEGER NOT NULL,
    content VARCHAR(1000) NOT NULL,
    is_deleted BOOLEAN NOT NULL DEFAULT FALSE,
    CONSTRAINT uq_recipe_steps_recipe_step UNIQUE (recipe_id, step_no)
);

CREATE TABLE IF NOT EXISTS recipe_required_products (
    recipe_required_product_id BIGSERIAL PRIMARY KEY,
    recipe_id BIGINT NOT NULL REFERENCES recipes(recipe_id),
    product_id BIGINT NOT NULL REFERENCES products(product_id),
    CONSTRAINT uq_recipe_required_products_recipe_product UNIQUE (recipe_id, product_id)
);
