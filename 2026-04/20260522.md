## api서블릿 예시
// 이 서블릿이 /api로 시작하는 모든 요청을 받음
@WebServlet("/api/*")
public class ApiDispatcherServlet extends HttpServlet {

  private final ObjectMapper mapper = new ObjectMapper();
  private final MemoController  memo = new MemoController();

  // get, post같은 여러 http method 요청이 들어왔을 때 공통으로 실행되는 메소드
  @Override
  protected void service(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
      req.setCharacterEncoding("UTF-8");
      resp.setContentType("application/json; charset=UTF-8");

      String method = req.getMethod();
      String path = req.getPathInfo();

      System.out.println(method + path);

      if (path == null) {
          sendError(resp, HttpServletResponse.SC_NOT_FOUND, "잘못된 API 경로입니다.");
          return;
      }

      if ("GET".equals(method)&& "/memos".equals(path)) {
          memo.findAll(resp);
          return;
      }
      if ("POST".equals(method) && "/memos".equals(path)) {
          memo.regist(req, resp);
          return;
      }
      if ("POST".equals(method) && "/memos/delete".equals(path)) {
          memo.remove(req, resp);
          return;
      }

      sendError(resp, HttpServletResponse.SC_NOT_FOUND, "잘못된 API 경로입니다.");

  }
